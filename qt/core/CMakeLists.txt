# The library core, linked into the application, and room237-core, the update helper without Qt.
# Third-party C sources come from qt/packaging/ci/deps.lock.json (group "core-sources"), checked by size and SHA-256.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  # Standalone build of the core and its unit tests: cmake -S qt/core -DROOM237_CORE_TESTS=ON.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ROOM237_VERSION)
  string(STRIP "${ROOM237_VERSION}" ROOM237_VERSION)
  if(APPLE)
    set(_languages C CXX OBJCXX)
  else()
    set(_languages C CXX)
  endif()
  project(room237_core VERSION 0.0.0 LANGUAGES ${_languages})
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_AUTOMOC ON)
  option(ROOM237_QA "Build the QA hooks" OFF)
endif()
enable_language(C)
option(ROOM237_CORE_TESTS "Build the core unit tests" OFF)

set(ROOM237_CORE_SOURCES "" CACHE PATH "Directory with the unpacked core-sources group (fetched at configure time when empty)")
if(NOT ROOM237_CORE_SOURCES)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(_deps "${CMAKE_BINARY_DIR}/core-deps")
  set(_cache "$ENV{ROOM237_DEPS_CACHE}")
  if(NOT _cache)
    set(_cache "${_deps}/downloads")
  endif()
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../packaging/ci/fetch_deps.py" --dest "${_deps}" --cache "${_cache}" core-sources
    RESULT_VARIABLE _fetched)
  if(NOT _fetched EQUAL 0)
    message(FATAL_ERROR "Fetching the pinned core sources failed")
  endif()
  set(ROOM237_CORE_SOURCES "${_deps}/core-sources" CACHE PATH "" FORCE)
endif()
set(_third "${ROOM237_CORE_SOURCES}")
foreach(_dir sqlite libwebp monocypher zlib)
  if(NOT EXISTS "${_third}/${_dir}")
    message(FATAL_ERROR "${_third}/${_dir} is missing; run qt/packaging/ci/fetch_deps.py core-sources")
  endif()
endforeach()

# SQLite with the options the earlier bundled build used.
add_library(room237_sqlite STATIC "${_third}/sqlite/sqlite3.c")
target_include_directories(room237_sqlite SYSTEM PUBLIC "${_third}/sqlite")
target_compile_definitions(room237_sqlite PRIVATE
  SQLITE_CORE SQLITE_DEFAULT_FOREIGN_KEYS=1 SQLITE_ENABLE_API_ARMOR SQLITE_ENABLE_COLUMN_METADATA SQLITE_ENABLE_DBSTAT_VTAB
  SQLITE_ENABLE_FTS3 SQLITE_ENABLE_FTS3_PARENTHESIS SQLITE_ENABLE_FTS5 SQLITE_ENABLE_JSON1 SQLITE_OMIT_LOAD_EXTENSION
  SQLITE_ENABLE_MEMORY_MANAGEMENT SQLITE_ENABLE_RTREE SQLITE_ENABLE_STAT4 SQLITE_SOUNDEX SQLITE_THREADSAFE=1 SQLITE_USE_URI
  HAVE_USLEEP=1 HAVE_ISNAN _POSIX_THREAD_SAFE_FUNCTIONS)
set_target_properties(room237_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)

# libwebp: decoder, encoder and sharpyuv, as the earlier libwebp-sys build compiled them.
file(GLOB _webp_sources "${_third}/libwebp/src/dec/*.c" "${_third}/libwebp/src/dsp/*.c" "${_third}/libwebp/src/enc/*.c" "${_third}/libwebp/src/utils/*.c" "${_third}/libwebp/sharpyuv/*.c")
add_library(room237_webp STATIC ${_webp_sources})
target_include_directories(room237_webp SYSTEM PUBLIC "${_third}/libwebp/src" PRIVATE "${_third}/libwebp")
target_compile_definitions(room237_webp PRIVATE NDEBUG=1 _THREAD_SAFE=1)
if(NOT WIN32)
  target_compile_definitions(room237_webp PRIVATE WEBP_USE_THREAD=1)
endif()
set_target_properties(room237_webp PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)

add_library(room237_monocypher STATIC "${_third}/monocypher/src/monocypher.c" "${_third}/monocypher/src/optional/monocypher-ed25519.c")
target_include_directories(room237_monocypher SYSTEM PUBLIC "${_third}/monocypher/src" "${_third}/monocypher/src/optional")
set_target_properties(room237_monocypher PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
# Argon2id runs at every unlock; unoptimized, it takes ten times longer.
if(NOT MSVC)
  target_compile_options(room237_monocypher PRIVATE $<$<CONFIG:Debug>:-O2>)
endif()

set(_zlib_names adler32 compress crc32 deflate gzclose gzlib gzread gzwrite inflate infback inftrees inffast trees uncompr zutil)
set(_zlib_sources "")
foreach(_name ${_zlib_names})
  list(APPEND _zlib_sources "${_third}/zlib/${_name}.c")
endforeach()
add_library(room237_zlib STATIC ${_zlib_sources})
target_include_directories(room237_zlib SYSTEM PUBLIC "${_third}/zlib")
if(NOT WIN32)
  target_compile_definitions(room237_zlib PRIVATE HAVE_UNISTD_H)
endif()
set_target_properties(room237_zlib PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Concurrent)
if(UNIX AND NOT APPLE)
  find_package(Threads REQUIRED)
endif()
set(ROOM237_CORE_SRC
  src/album.cpp src/album_db.cpp src/album_tree.cpp src/catalog.cpp src/clipboard.cpp src/constants.cpp src/db.cpp
  src/disk_duplicates.cpp src/duplicates.cpp src/entry.cpp src/fs.cpp src/host.cpp src/image_hash.cpp src/json.cpp
  src/maintenance.cpp src/media_delete.cpp src/media_move.cpp src/media_probe.cpp src/metadata.cpp src/preload.cpp
  src/protocol.cpp src/settings.cpp src/thumb.cpp src/update.cpp src/util.cpp src/vault.cpp src/vault_ops.cpp src/factors.cpp src/factor_other.cpp src/webp.cpp src/xattr.cpp)
if(WIN32)
  list(APPEND ROOM237_CORE_SRC src/factor_windows.cpp)
endif()
# TPM 2.0 on Linux: tpm2-tss headers at build time; the libraries are loaded at run time when present.
if(UNIX AND NOT APPLE)
  option(ROOM237_TPM "TPM 2.0 with a PIN as a second factor" ON)
  if(ROOM237_TPM)
    include(CheckIncludeFileCXX)
    check_include_file_cxx(tss2/tss2_esys.h ROOM237_HAVE_TSS2)
    if(NOT ROOM237_HAVE_TSS2)
      message(FATAL_ERROR "tss2/tss2_esys.h is missing: install libtss2-dev, or configure with -DROOM237_TPM=OFF")
    endif()
    list(APPEND ROOM237_CORE_SRC src/factor_tpm.cpp)
  endif()
endif()
if(APPLE)
  list(APPEND ROOM237_CORE_SRC src/native_image_mac.mm src/factor_mac.cpp)
  # The Secure Enclave factor uses CryptoKit, which has only a Swift interface.
  enable_language(Swift)
  add_library(room237_se STATIC src/factor_se.swift)
  set_target_properties(room237_se PROPERTIES Swift_MODULE_NAME Room237SecureEnclave)
  target_link_libraries(room237_se INTERFACE "-framework CryptoKit" "-framework LocalAuthentication" "-framework Security")
endif()
option(ROOM237_FIDO "Security keys (FIDO2) as a second factor" ON)
if(ROOM237_FIDO)
  include(cmake/fido.cmake)
  list(APPEND ROOM237_CORE_SRC src/factor_fido.cpp)
endif()
add_library(room237_core STATIC ${ROOM237_CORE_SRC})
target_include_directories(room237_core PUBLIC src)
set_target_properties(room237_core PROPERTIES LINKER_LANGUAGE CXX)
target_compile_features(room237_core PUBLIC cxx_std_20)
# Perceptual hashes must round like the earlier implementation, which never fused multiply-adds.
target_compile_options(room237_core PRIVATE $<$<OR:$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>,$<CXX_COMPILER_ID:GNU>>:-ffp-contract=off>)
target_compile_definitions(room237_core PRIVATE ROOM237_VERSION="${ROOM237_VERSION}" QT_NO_KEYWORDS)
if(ROOM237_UPDATE_PUBKEY)
  target_compile_definitions(room237_core PRIVATE ROOM237_UPDATE_PUBKEY="${ROOM237_UPDATE_PUBKEY}")
endif()
if(ROOM237_QA)
  target_compile_definitions(room237_core PUBLIC ROOM237_QA)
endif()
target_link_libraries(room237_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent PRIVATE room237_sqlite room237_webp room237_monocypher room237_zlib)
if(ROOM237_FIDO)
  target_link_libraries(room237_core PRIVATE room237_fido)
endif()
if(WIN32)
  target_link_libraries(room237_core PRIVATE runtimeobject)
endif()
if(UNIX AND NOT APPLE AND ROOM237_TPM)
  target_compile_definitions(room237_core PRIVATE ROOM237_TPM)
endif()
if(APPLE)
  target_link_libraries(room237_core PRIVATE "-framework CoreFoundation" "-framework CoreGraphics" "-framework ImageIO" room237_se)
elseif(UNIX)
  target_link_libraries(room237_core PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
endif()

# The helper keeps the name of the earlier core binary: an older app verifies a staged install with
# "room237-core --version" before it swaps it in.
add_executable(room237-core src/update_helper.cpp)
target_compile_features(room237-core PRIVATE cxx_std_20)
target_compile_definitions(room237-core PRIVATE ROOM237_VERSION="${ROOM237_VERSION}")
if(ROOM237_QA)
  target_compile_definitions(room237-core PRIVATE ROOM237_QA)
endif()
if(WIN32)
  target_link_options(room237-core PRIVATE -municode)
endif()

if(ROOM237_CORE_TESTS)
  if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "The core unit tests build the core with a test factor; configure them from qt/core alone")
  endif()
  target_compile_definitions(room237_core PRIVATE ROOM237_TEST_FACTOR)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  enable_testing()
  add_executable(room237_core_tests tests/core_tests.cpp)
  set_target_properties(room237_core_tests PROPERTIES AUTOMOC ON)
  target_compile_definitions(room237_core_tests PRIVATE ROOM237_TESTDATA="${CMAKE_CURRENT_SOURCE_DIR}/testdata")
  target_link_libraries(room237_core_tests PRIVATE room237_core room237_zlib Qt6::Test)
  add_test(NAME room237_core_tests COMMAND room237_core_tests)
endif()
