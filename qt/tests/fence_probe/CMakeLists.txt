cmake_minimum_required(VERSION 3.21)
project(room237_fence_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Widgets)
if(NOT DEFINED ROOM237_SOURCE_ROOT)
  message(FATAL_ERROR "ROOM237_SOURCE_ROOT must point to the directory that contains qt/")
endif()
set(S "${ROOM237_SOURCE_ROOT}/qt/app/src")
add_executable(qa_fence_probe ../qa_fence_probe.cpp
  "${S}/qa_fence.cpp" "${S}/updater.cpp" "${S}/updater.h"
  "${S}/core_client.cpp" "${S}/core_client.h" "${S}/qatrace.cpp"
  "${S}/i18n.cpp" "${S}/platform.cpp")
target_include_directories(qa_fence_probe PRIVATE "${S}")
file(READ "${ROOM237_SOURCE_ROOT}/qt/VERSION" ROOM237_VERSION)
string(STRIP "${ROOM237_VERSION}" ROOM237_VERSION)
target_compile_definitions(qa_fence_probe PRIVATE ROOM237_VERSION="${ROOM237_VERSION}" ROOM237_QA)
target_link_libraries(qa_fence_probe PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(qa_fence_probe PRIVATE "${S}/platform_mac.mm")
  target_link_libraries(qa_fence_probe PRIVATE "-framework AppKit" "-framework Foundation" "-framework CoreGraphics" "-framework ImageIO" "-framework CoreServices")
elseif(UNIX)
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  target_link_libraries(qa_fence_probe PRIVATE Qt6::DBus)
endif()
